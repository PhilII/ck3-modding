# Shown when the rules for this family member have been relaxed
dap_cl_member_relaxed = {
	type = character
	text = {
		trigger = {
			has_variable = dap_relax
			var:dap_relax >= 1
		}
		localization_key = dap_relaxed_note
	}
	text = {
		fallback = yes
		localization_key = dap_empty
	}
}

# Guardian line in the Guardians tab
dap_cl_guardian = {
	type = character
	text = {
		trigger = { has_variable = dap_excluded }
		localization_key = dap_guardian_excluded
	}
	text = {
		trigger = { is_adult = yes }
		localization_key = dap_guardian_adult
	}
	text = {
		trigger = { num_of_relation_guardian = 0 }
		localization_key = dap_guardian_none
	}
	text = {
		trigger = {
			has_variable = dap_guardian
			any_relation = {
				type = guardian
				this = prev.var:dap_guardian
			}
		}
		localization_key = dap_guardian_by_autopilot
	}
	text = {
		fallback = yes
		localization_key = dap_guardian_has
	}
}
