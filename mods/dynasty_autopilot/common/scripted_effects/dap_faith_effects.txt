# Dynasty Autopilot - faith module: court chaplain, conversion, piety and Head of Faith.
# root = the player in every effect here.

### Court Chaplain

# this = player; fill or improve the chaplain seat, unless the clergy appoints it
dap_run_chaplain_seat_effect = {
	if = {
		limit = {
			var:dap_faith_chaplain >= 1
			tgp_has_access_to_ministry_trigger = no
			can_appoint_own_court_chaplain_trigger = yes
		}
		dap_fill_council_seat_effect = { SEAT = councillor_court_chaplain CAN_BE = can_be_court_chaplain_trigger SKILL = learning MODE = dap_faith_chaplain }
	}
}

# this = player; set the chaplain to convert the next county, never interrupting a conversion
dap_run_chaplain_task_effect = {
	if = {
		limit = {
			var:dap_faith_task = 1
			exists = cp:councillor_court_chaplain
			cp:councillor_court_chaplain = {
				faith = root.faith
				NOT = { is_performing_council_task = task_conversion }
			}
		}
		cp:councillor_court_chaplain = { save_scope_as = dap_chaplain }
		every_realm_county = {
			limit = { dap_convertible_county_trigger = yes }
			add_to_list = dap_convert_pool
		}
		ordered_in_list = {
			list = dap_convert_pool
			order_by = dap_convert_county_score
			max = 1
			check_range_bounds = no
			save_scope_as = dap_convert_county
			if = {
				limit = { dap_dry_run_trigger = no }
				scope:dap_chaplain = {
					set_council_task = {
						task_type = task_conversion
						target = scope:dap_convert_county.title_province
					}
				}
			}
			scope:dap_chaplain = {
				dap_log_effect = { ACTION = dap_log_convert_county }
				set_variable = { name = dap_log_county value = scope:dap_convert_county }
			}
		}
	}
}

### Family

# this = courtier of root; ask them to adopt root's faith if they would accept
dap_ask_conversion_effect = {
	save_scope_as = dap_convertee
	if = {
		limit = {
			is_ruler = no
			is_courtier_of = root
			faith != root.faith
			root = {
				is_character_interaction_potentially_accepted = {
					recipient = scope:dap_convertee
					interaction = ask_for_conversion_courtier_interaction
					required_response = yes
				}
			}
		}
		# run_interaction does nothing for this interaction (verified in 1.20.1.2), so run its accept effect
		if = {
			limit = { dap_dry_run_trigger = no }
			root = { save_scope_as = actor }
			scope:dap_convertee = { save_scope_as = recipient }
			demand_conversion_interaction_effect = yes
		}
		dap_log_effect = { ACTION = dap_log_converted }
	}
}

dap_run_family_faith_effect = {
	if = {
		limit = { var:dap_faith_family = 1 }
		every_courtier = {
			limit = {
				dap_is_family_member_trigger = yes
				NOT = { has_variable = dap_excluded }
			}
			dap_ask_conversion_effect = yes
		}
	}
	if = {
		limit = { var:dap_faith_spouses = 1 }
		every_spouse = {
			dap_ask_conversion_effect = yes
		}
		every_concubine = {
			dap_ask_conversion_effect = yes
		}
	}
}

# this = vassal ruler of root; ask them to adopt root's faith, with family and capital, if they would accept
dap_ask_vassal_conversion_effect = {
	save_scope_as = dap_convertee
	if = {
		limit = {
			faith != root.faith
			root = {
				is_character_interaction_potentially_accepted = {
					recipient = scope:dap_convertee
					interaction = demand_conversion_vassal_ruler_interaction
					required_response = yes
				}
			}
		}
		# As for courtiers, run the interaction's accept effect directly
		if = {
			limit = { dap_dry_run_trigger = no }
			root = { save_scope_as = actor }
			scope:dap_convertee = { save_scope_as = recipient }
			demand_conversion_vassal_ruler_interaction_effect = yes
		}
		dap_log_effect = { ACTION = dap_log_converted }
	}
}

dap_run_vassal_faith_effect = {
	if = {
		limit = { var:dap_faith_vassals >= 1 }
		every_vassal = {
			limit = {
				is_ai = yes
				trigger_if = {
					limit = { root.var:dap_faith_vassals = 1 }
					dap_is_kin_trigger = yes
				}
			}
			dap_ask_vassal_conversion_effect = yes
		}
	}
}

### Piety and Head of Faith

# this = player; ask the Head of Faith for gold if they would agree and piety is above the reserve
dap_ask_hof_gold_effect = {
	if = {
		limit = {
			var:dap_hof_gold = 1
			NOT = { has_variable = dap_hof_gold_cooldown }
			exists = faith.religious_head
			NOT = { faith.religious_head = root }
			piety >= {
				value = dap_piety_reserve_value
				add = major_piety_value
			}
		}
		faith.religious_head = { save_scope_as = dap_hof }
		if = {
			limit = {
				is_character_interaction_potentially_accepted = {
					recipient = scope:dap_hof
					interaction = hof_ask_for_gold_interaction
					required_response = yes
				}
			}
			if = {
				limit = { dap_dry_run_trigger = no }
				save_scope_as = actor
				scope:dap_hof = { save_scope_as = recipient }
				save_scope_value_as = { name = hook value = no }
				hof_ask_for_gold_interaction_effect = yes
				set_variable = { name = dap_hof_gold_cooldown value = 1 years = 3 }
			}
			scope:dap_hof = { dap_log_effect = { ACTION = dap_log_hof_gold } }
		}
	}
}

# this = player; one toast per excommunication
dap_excomm_alert_effect = {
	if = {
		limit = {
			has_trait = excommunicated
			var:dap_excomm_alert = 1
			NOT = { has_variable = dap_excomm_warned }
		}
		set_variable = dap_excomm_warned
		send_interface_toast = {
			type = event_toast_effect_bad
			title = dap_excomm_title
			desc = dap_excomm_desc
		}
	}
	else_if = {
		limit = {
			has_variable = dap_excomm_warned
			NOT = { has_trait = excommunicated }
		}
		remove_variable = dap_excomm_warned
	}
}

dap_run_faith_effect = {
	if = {
		limit = { var:dap_faith_enabled = 1 }
		dap_run_chaplain_seat_effect = yes
		dap_run_chaplain_task_effect = yes
		dap_run_family_faith_effect = yes
		dap_run_vassal_faith_effect = yes
		dap_ask_hof_gold_effect = yes
		dap_excomm_alert_effect = yes
	}
}
