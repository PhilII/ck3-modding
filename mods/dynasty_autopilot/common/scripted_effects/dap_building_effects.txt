# Dynasty Autopilot - automatic building through the real Build button (ADR 0001).
# root = the player in every effect here.
#
# Route: once a season the builder scores every new building and upgrade in the holdings it
# may build in (dap_build_holdings: the player's own, and their vassal barons', cities' and
# temples' holdings in the player's own counties; military bases, dap_military_bases, are
# among the player's own; with the vassals' counties option, the holdings in the counties of
# their direct vassals once their own have nothing left, dap_build_vassal_holdings), and queues the best (dap_build_queue, a list of provinces, each
# holding its pick in var:dap_bc_*). gui/dap_build_driver.gui asks for them one at a time (the
# dap_build_next scripted GUI): the mod remembers which building it wants in which holding
# and opens the vanilla holding view for that holding. gui/00_dap_building_item.gui (an
# override of the holding view's building slot type) presses Build, or Upgrade, on that
# building, then calls the dap_build_started scripted GUI, which runs
# dap_build_started_effect. Only the driver opens the view, so it never opens twice at once.

# Asks for a construction of var:dap_build_building (flag:<building key>) in
# scope:dap_build_province, as an upgrade of var:dap_build_from (flag:<building key>, or
# flag:none for a new building in a free slot), both set for a day, and opens the holding
# view so it can be pressed. The request lapses after a day, so one that could not be
# pressed never fires later, when the player opens that holding's view themselves.
dap_build_open_effect = {
	if = {
		limit = { has_variable = dap_build_last_province }
		remove_variable = dap_build_last_province
		remove_variable = dap_build_last_paid
	}
	set_variable = { name = dap_build_province value = scope:dap_build_province days = 1 }
	set_variable = { name = dap_build_gold_before value = gold days = 1 }
	# Whose turn it is among the two Build presses and the two Upgrade presses of
	# gui/00_dap_building_item.gui (dap_build_pending_a: 0, _b: 1); see ADR 0001
	if = {
		limit = { var:dap_build_turn ?= 1 }
		set_variable = { name = dap_build_turn value = 0 }
	}
	else = { set_variable = { name = dap_build_turn value = 1 } }
	scope:dap_build_province = { open_view_data = holding_view }
}

# Asks for the most valuable queued construction, if there is one. A request the holding
# view cannot press (the building can no longer be built there) lapses after a day, and
# the next one goes.
dap_build_next_effect = {
	if = {
		limit = { any_in_list = { variable = dap_build_queue } }
		ordered_in_list = {
			variable = dap_build_queue
			order_by = var:dap_bc_value
			save_scope_as = dap_build_province
		}
		remove_list_variable = { name = dap_build_queue target = scope:dap_build_province }
		set_variable = { name = dap_build_building value = scope:dap_build_province.var:dap_bc_building days = 1 }
		set_variable = { name = dap_build_from value = scope:dap_build_province.var:dap_bc_from days = 1 }
		dap_build_open_effect = yes
	}
}

# The holding view pressed Build for the request in scope:province; the last
# construction's holding and price stay until the next request
dap_build_started_effect = {
	set_variable = { name = dap_build_last_province value = scope:province }
	set_variable = { name = dap_build_last_paid value = var:dap_build_gold_before }
	change_variable = { name = dap_build_last_paid subtract = gold }
	dap_log_effect = { ACTION = dap_log_built }
	set_variable = { name = dap_log_building value = var:dap_build_building }
	set_variable = { name = dap_log_province value = scope:province }
	scope:province = { remove_variable = dap_build_queued }
	remove_variable = dap_build_province
	remove_variable = dap_build_building
	remove_variable = dap_build_from
	remove_variable = dap_build_gold_before
}

# this = province; forgets its last pick
dap_clear_building_pick_effect = {
	if = {
		limit = { has_variable = dap_bc_value }
		remove_variable = dap_bc_value
		remove_variable = dap_bc_building
		remove_variable = dap_bc_from
		remove_variable = dap_bc_cost
	}
}

# this = province; picks the building worth the most for its cost there, new or upgrade,
# into var:dap_bc_building (flag:<key>), var:dap_bc_from (flag:<key upgraded> or
# flag:none), var:dap_bc_cost and var:dap_bc_value; no pick if nothing there serves an
# active focus. Needs local_var:dap_w_income, local_var:dap_w_development and
# local_var:dap_w_military, scope:dap_stack_income and scope:dap_stack_development (the
# stacking county of each focus, if it has one), and the list dap_military_bases.
dap_pick_building_in_province_effect = {
	dap_clear_building_pick_effect = yes
	province_owner = { save_scope_as = holder }
	county = { save_scope_as = county }
	set_local_variable = { name = dap_best_value value = 0 }
	# What each reach of bonus is worth to the player here: holding and holder bonuses in
	# full in their own holding; in a vassal's, the holding's bonus only as the tax the
	# vassal pays them, and the holder's not at all
	if = {
		limit = { province_owner = root }
		set_local_variable = { name = dap_share_holding value = 1 }
		set_local_variable = { name = dap_share_holder value = 1 }
	}
	else = {
		scope:holder = {
			set_local_variable = { name = dap_share_holding value = dap_vassal_tax_share_value }
			# Two levels down, the tax reaches the player through the holder's liege
			if = {
				limit = { NOT = { is_vassal_of = root } }
				liege = { change_local_variable = { name = dap_share_holding multiply = dap_vassal_tax_share_value } }
			}
		}
		set_local_variable = { name = dap_share_holder value = 0 }
	}
	# County stacking: county-wide bonuses count only in the focus's stacking county
	set_local_variable = { name = dap_stack_income value = 0 }
	if = {
		limit = { scope:dap_stack_income ?= scope:county }
		set_local_variable = { name = dap_stack_income value = 1 }
	}
	set_local_variable = { name = dap_stack_development value = 0 }
	if = {
		limit = { scope:dap_stack_development ?= scope:county }
		set_local_variable = { name = dap_stack_development value = 1 }
	}
	# Military stacking: stationed bonuses count only in a military base, for its specialty
	set_local_variable = { name = dap_base_here value = 0 }
	save_temporary_scope_as = dap_here
	if = {
		limit = { root = { is_target_in_variable_list = { name = dap_military_bases target = scope:dap_here } } }
		set_local_variable = { name = dap_base_here value = 1 }
	}
	dap_consider_buildings_effect = yes
}

# this = province; one candidate offered by dap_consider_buildings_effect (generated by
# tools/build_buildings.py). Its value is what it gives the active focuses per 1000 gold,
# each bonus as much as it reaches the player (dap_pick_building_in_province_effect).
# With local_var:dap_room_pass set it is only noted whether a county-wide bonus of a focus
# could still be stacked in this county (dap_build_room_income, dap_build_room_development);
# with local_var:dap_base_room_pass, whether this military base could still stack its
# specialty (var:dap_base_room).
dap_consider_building_effect = {
	if = {
		limit = { exists = local_var:dap_room_pass }
		set_local_variable = { name = dap_bv value = $INCOME_COUNTY$ }
		if = {
			limit = {
				local_var:dap_bv > 0
				root = { NOT = { is_target_in_variable_list = { name = dap_build_room_income target = scope:county } } }
			}
			root = { add_to_variable_list = { name = dap_build_room_income target = scope:county } }
		}
		set_local_variable = { name = dap_bv value = $DEVELOPMENT_COUNTY$ }
		if = {
			limit = {
				local_var:dap_bv > 0
				root = { NOT = { is_target_in_variable_list = { name = dap_build_room_development target = scope:county } } }
			}
			root = { add_to_variable_list = { name = dap_build_room_development target = scope:county } }
		}
	}
	else_if = {
		limit = { exists = local_var:dap_base_room_pass }
		set_local_variable = { name = dap_bv value = $MILITARY$ }
		if = {
			limit = { local_var:dap_bv > 0 }
			set_variable = { name = dap_base_room value = 1 }
		}
	}
	else = {
		dap_score_building_effect = {
			BUILDING = $BUILDING$ FROM = $FROM$ COST = $COST$ INCOME_HOLDING = $INCOME_HOLDING$
			INCOME_HOLDER = $INCOME_HOLDER$ INCOME_COUNTY = $INCOME_COUNTY$ DEVELOPMENT_COUNTY = $DEVELOPMENT_COUNTY$
			MILITARY = $MILITARY$
		}
	}
}

# this = province; scores one candidate and keeps it as the pick if it beats the best so far
dap_score_building_effect = {
	set_local_variable = {
		name = dap_bv
		value = {
			value = $INCOME_HOLDING$
			multiply = local_var:dap_share_holding
			add = {
				value = $INCOME_HOLDER$
				multiply = local_var:dap_share_holder
			}
			add = {
				value = $INCOME_COUNTY$
				multiply = local_var:dap_stack_income
			}
			multiply = local_var:dap_w_income
			add = {
				value = $DEVELOPMENT_COUNTY$
				multiply = local_var:dap_stack_development
				multiply = local_var:dap_w_development
			}
			add = {
				value = $MILITARY$
				multiply = local_var:dap_base_here
				multiply = local_var:dap_w_military
			}
			multiply = 1000
			divide = $COST$
		}
	}
	if = {
		limit = { local_var:dap_bv > local_var:dap_best_value }
		set_local_variable = { name = dap_best_value value = local_var:dap_bv }
		set_variable = { name = dap_bc_value value = local_var:dap_bv }
		set_variable = { name = dap_bc_building value = flag:$BUILDING$ }
		set_variable = { name = dap_bc_from value = flag:$FROM$ }
		set_variable = { name = dap_bc_cost value = $COST$ }
	}
}

# this = player; the weight of the focus setting var:$SETTING$ (Off/Low/High = 0/1/3) as
# local_var:$WEIGHT$
dap_focus_weight_effect = {
	set_local_variable = { name = $WEIGHT$ value = var:$SETTING$ }
	if = {
		limit = { var:$SETTING$ = 2 }
		set_local_variable = { name = $WEIGHT$ value = 3 }
	}
}

# this = player; focus weights as local variables
dap_building_weights_effect = {
	dap_focus_weight_effect = { SETTING = dap_build_income WEIGHT = dap_w_income }
	dap_focus_weight_effect = { SETTING = dap_build_development WEIGHT = dap_w_development }
	dap_focus_weight_effect = { SETTING = dap_build_military WEIGHT = dap_w_military }
}

# this = province, root = player; adds it to the list dap_listed_holdings, marking it var:dap_listed_first if
# it is the first listed of its county
dap_add_listed_holding_effect = {
	save_temporary_scope_as = dap_listed
	if = {
		limit = { root = { NOT = { any_in_list = { variable = dap_listed_holdings county = scope:dap_listed.county } } } }
		set_variable = { name = dap_listed_first value = 1 }
	}
	root = { add_to_variable_list = { name = dap_listed_holdings target = scope:dap_listed } }
}

# this = player; every holding the builder could build in, county by county, into the list
# dap_listed_holdings, which the window lists with exclude buttons: the player's own, and their
# vassals' (barons, cities, temples) in the player's own counties; with the vassals' counties
# option, every holding in the counties of the player's direct vassals, theirs and their own
# vassals' (a liege two levels up may build there, verified in 1.20.1.2)
dap_list_window_holdings_effect = {
	every_in_list = {
		variable = dap_listed_holdings
		limit = { has_variable = dap_listed_first }
		remove_variable = dap_listed_first
	}
	clear_variable_list = dap_listed_holdings
	every_held_title = {
		limit = { tier = tier_county }
		every_county_province = {
			limit = {
				has_holding = yes
				OR = {
					province_owner = root
					province_owner = { is_vassal_of = root }
				}
			}
			dap_add_listed_holding_effect = yes
		}
	}
	# Holdings of the player's own in counties someone else holds, grouped by county; a direct
	# vassal's county comes in full below when the vassals' counties option is on
	every_directly_owned_province = {
		limit = {
			has_holding = yes
			NOT = {
				AND = {
					root = { var:dap_build_vassal_counties = 1 }
					county.holder = { is_vassal_of = root }
				}
			}
		}
		save_temporary_scope_as = dap_owned
		if = {
			limit = { root = { NOT = { is_target_in_variable_list = { name = dap_listed_holdings target = scope:dap_owned } } } }
			county = {
				every_county_province = {
					limit = {
						has_holding = yes
						province_owner = root
					}
					dap_add_listed_holding_effect = yes
				}
			}
		}
	}
	if = {
		limit = { var:dap_build_vassal_counties = 1 }
		every_vassal = {
			save_temporary_scope_as = dap_vassal
			every_held_title = {
				limit = { tier = tier_county }
				every_county_province = {
					limit = {
						has_holding = yes
						OR = {
							province_owner = root
							province_owner = scope:dap_vassal
							province_owner = { is_vassal_of = scope:dap_vassal }
						}
					}
					dap_add_listed_holding_effect = yes
				}
			}
		}
	}
}

# this = player; the holdings the builder may build in: those of
# dap_list_window_holdings_effect that are not excluded holdings, into the list
# dap_build_holdings, but those in vassals' counties into dap_build_vassal_holdings
dap_list_build_holdings_effect = {
	dap_list_window_holdings_effect = yes
	clear_variable_list = dap_build_holdings
	clear_variable_list = dap_build_vassal_holdings
	every_in_list = {
		variable = dap_listed_holdings
		limit = { dap_build_excluded_trigger = no }
		if = {
			limit = { dap_in_vassal_county_trigger = yes }
			root = { add_to_variable_list = { name = dap_build_vassal_holdings target = prev } }
		}
		else = {
			root = { add_to_variable_list = { name = dap_build_holdings target = prev } }
		}
	}
}

# this = province, root = player; drops the construction it waits for, if any, queued or
# already asked of the holding view, and its pick
dap_drop_build_request_effect = {
	save_temporary_scope_as = dap_dropped
	if = {
		limit = { has_variable = dap_build_queued }
		remove_variable = dap_build_queued
		root = { remove_list_variable = { name = dap_build_queue target = scope:dap_dropped } }
	}
	root = {
		if = {
			limit = {
				has_variable = dap_build_building
				var:dap_build_province ?= scope:dap_dropped
			}
			remove_variable = dap_build_province
			remove_variable = dap_build_building
			remove_variable = dap_build_from
			remove_variable = dap_build_gold_before
		}
	}
	dap_clear_building_pick_effect = yes
}

# this = province, root = player; takes the holding out of automatic building (an excluded holding), or
# puts it back. The flag lives on the province, so it survives succession
dap_toggle_build_excluded_effect = {
	if = {
		limit = { has_variable = dap_build_excluded }
		remove_variable = dap_build_excluded
	}
	else = {
		set_variable = { name = dap_build_excluded value = 1 }
		dap_drop_build_request_effect = yes
	}
}

# this = county title, root = player; takes every holding of the county out of automatic building, or puts
# them back. The flag lives on the title, so it survives succession
dap_toggle_county_build_excluded_effect = {
	if = {
		limit = { has_variable = dap_build_excluded }
		remove_variable = dap_build_excluded
	}
	else = {
		set_variable = { name = dap_build_excluded value = 1 }
		every_county_province = { dap_drop_build_request_effect = yes }
	}
}

# this = province; no longer a military base the player set
dap_clear_specialty_effect = {
	if = {
		limit = { has_variable = dap_specialty }
		remove_variable = dap_specialty
	}
	if = {
		limit = { has_variable = dap_specialty_set }
		remove_variable = dap_specialty_set
	}
}

# this = player; while Military stacking is on, the military bases into the list
# dap_military_bases: holdings of the player's domain, each stacking the stationed bonuses
# of one men-at-arms base type the player owns (its specialty, var:dap_specialty). Bases the
# player set (var:dap_specialty_set) come first, then dap_military_base_for_type_effect for
# every owned type; a holding forgets an automatic specialty that no base uses any more.
# Excluded holdings are never bases. Specialties live on the holding, so they stay with it
# through succession.
dap_update_military_bases_effect = {
	clear_variable_list = dap_military_bases
	if = {
		limit = { var:dap_build_military >= 1 }
		# Whether each earlier base still has room: a free slot, or an upgrade that stacks its
		# specialty
		every_directly_owned_province = {
			limit = { has_variable = dap_base_room }
			remove_variable = dap_base_room
		}
		set_local_variable = { name = dap_base_room_pass value = 1 }
		every_directly_owned_province = {
			limit = {
				has_variable = dap_specialty
				has_holding = yes
			}
			if = {
				limit = { has_free_building_slot = yes }
				set_variable = { name = dap_base_room value = 1 }
			}
			else = {
				province_owner = { save_scope_as = holder }
				county = { save_scope_as = county }
				dap_consider_buildings_effect = yes
			}
		}
		remove_local_variable = dap_base_room_pass
		every_directly_owned_province = {
			limit = {
				has_holding = yes
				has_variable = dap_specialty_set
				dap_owns_specialty_trigger = yes
				dap_build_excluded_trigger = no
			}
			root = { add_to_variable_list = { name = dap_military_bases target = prev } }
		}
		dap_military_bases_for_owned_types_effect = yes
		every_directly_owned_province = {
			limit = {
				has_variable = dap_specialty
				NOT = { has_variable = dap_specialty_set }
			}
			save_temporary_scope_as = dap_here
			if = {
				limit = { root = { NOT = { is_target_in_variable_list = { name = dap_military_bases target = scope:dap_here } } } }
				remove_variable = dap_specialty
			}
		}
	}
}

# this = player; the military bases of base type $TYPE$, which the player owns: those the
# player set to it, else the automatic ones from earlier runs; and a new one when there is
# none yet, or when none has room left (no free slot, and no upgrade that stacks the type).
# The new base is the holding of the domain, not yet a base, whose buildings already give
# the type the most stationed bonuses, then whose terrain suits the type best; a holding
# whose buildings suit another owned type better, or with no free slot, only as a last resort.
dap_military_base_for_type_effect = {
	if = {
		limit = {
			NOT = { any_in_list = { variable = dap_military_bases var:dap_specialty ?= flag:$TYPE$ } }
		}
		every_directly_owned_province = {
			limit = {
				has_holding = yes
				NOT = { has_variable = dap_specialty_set }
				var:dap_specialty ?= flag:$TYPE$
				dap_build_excluded_trigger = no
			}
			root = { add_to_variable_list = { name = dap_military_bases target = prev } }
		}
	}
	if = {
		limit = {
			NOT = {
				any_in_list = {
					variable = dap_military_bases
					var:dap_specialty ?= flag:$TYPE$
					has_variable = dap_base_room
				}
			}
		}
		ordered_directly_owned_province = {
			limit = {
				has_holding = yes
				NOT = { has_variable = dap_specialty }
				dap_build_excluded_trigger = no
			}
			order_by = {
				value = dap_holding_$TYPE$_stationed_value
				multiply = 1000
				add = dap_terrain_fit_$TYPE$_value
				if = {
					limit = { dap_best_owned_stationed_value > dap_holding_$TYPE$_stationed_value }
					subtract = 100000
				}
				if = {
					limit = { has_free_building_slot = no }
					subtract = 1000000
				}
			}
			set_variable = { name = dap_specialty value = flag:$TYPE$ }
			root = { add_to_variable_list = { name = dap_military_bases target = prev } }
		}
	}
}

# this = player; county stacking: for each active focus with county-wide bonuses, saves
# its stacking county as scope:dap_stack_income or scope:dap_stack_development. That is
# the county, among those where such a bonus can still be built in some holding (busy ones
# included, so a county stays the stacking county while its constructions are under way),
# with the most buildings serving the focus already, the capital breaking ties.
dap_choose_stacking_counties_effect = {
	# Saved scopes last the whole event, so none is left from an earlier choice
	if = {
		limit = { exists = scope:dap_stack_income }
		clear_saved_scope = dap_stack_income
	}
	if = {
		limit = { exists = scope:dap_stack_development }
		clear_saved_scope = dap_stack_development
	}
	clear_variable_list = dap_build_room_income
	clear_variable_list = dap_build_room_development
	set_local_variable = { name = dap_room_pass value = 1 }
	every_in_list = {
		variable = dap_build_holdings
		province_owner = { save_scope_as = holder }
		county = { save_scope_as = county }
		dap_consider_buildings_effect = yes
	}
	remove_local_variable = dap_room_pass
	capital_county = { save_scope_as = dap_capital_county }
	if = {
		limit = { local_var:dap_w_income > 0 }
		ordered_in_list = {
			variable = dap_build_room_income
			order_by = dap_county_income_buildings_value
			save_scope_as = dap_stack_income
		}
	}
	if = {
		limit = { local_var:dap_w_development > 0 }
		ordered_in_list = {
			variable = dap_build_room_development
			order_by = dap_county_development_buildings_value
			save_scope_as = dap_stack_development
		}
	}
	clear_variable_list = dap_build_room_income
	clear_variable_list = dap_build_room_development
}

# this = player; picks the best building in every holding of dap_build_holdings that is
# neither waiting for a construction nor building one
dap_pick_buildings_effect = {
	every_in_list = {
		variable = dap_build_holdings
		limit = { NOT = { has_variable = dap_build_queued } }
		dap_clear_building_pick_effect = yes
		if = {
			limit = { has_ongoing_construction = no }
			dap_pick_building_in_province_effect = yes
		}
	}
}

# this = player; once a season, starts the new buildings and upgrades worth the most for
# their cost in the holdings it may build in (dap_list_build_holdings_effect), stacking
# county-wide bonuses in the stacking county (dap_choose_stacking_counties_effect) and
# stationed bonuses in military bases (dap_update_military_bases_effect), then, with the
# vassals' counties option and nothing left in their own counties, in their vassals' counties, within
# the gold above the reserve and the constructions allowed per season; in Suggest only mode
# (local_var:dap_dry) it logs them instead. A holding with construction under way, or one
# already waiting for the holding view, is skipped.
dap_run_buildings_effect = {
	if = {
		limit = {
			var:dap_build_enabled = 1
			OR = {
				var:dap_build_income >= 1
				var:dap_build_development >= 1
				var:dap_build_military >= 1
			}
		}
		# A queue whose marks lapsed (a day passed without the holding view pressing it) is stale
		if = {
			limit = {
				any_in_list = {
					variable = dap_build_queue
					NOT = { has_variable = dap_build_queued }
				}
			}
			every_in_list = {
				variable = dap_build_queue
				remove_variable = dap_build_queued
			}
			clear_variable_list = dap_build_queue
		}
		dap_building_weights_effect = yes
		# Constructions still allowed this season
		set_local_variable = {
			name = dap_build_left
			value = {
				value = var:dap_build_per_season
				add = 1
			}
		}
		if = {
			limit = { var:dap_build_per_season = 3 }
			set_local_variable = { name = dap_build_left value = 1000 }
		}
		if = {
			limit = { has_variable = dap_build_season_count }
			change_local_variable = { name = dap_build_left subtract = var:dap_build_season_count }
		}
		# Gold already promised to constructions waiting for the holding view
		set_local_variable = { name = dap_build_spent value = 0 }
		every_in_list = {
			variable = dap_build_queue
			change_local_variable = { name = dap_build_spent add = var:dap_bc_cost }
		}
		if = {
			limit = { has_variable = dap_build_building }
			var:dap_build_province = {
				if = {
					limit = { has_variable = dap_bc_cost }
					change_local_variable = { name = dap_build_spent add = var:dap_bc_cost }
				}
			}
		}
		dap_list_build_holdings_effect = yes
		dap_update_military_bases_effect = yes
		dap_choose_stacking_counties_effect = yes
		dap_pick_buildings_effect = yes
		# Vassals' counties, only once the player's own have nothing left: no pick, and no county
		# left to stack (holdings with construction under way included)
		if = {
			limit = {
				var:dap_build_vassal_counties = 1
				NOT = { exists = scope:dap_stack_income }
				NOT = { exists = scope:dap_stack_development }
				NOT = { any_in_list = { variable = dap_build_holdings has_variable = dap_bc_value } }
			}
			clear_variable_list = dap_build_holdings
			every_in_list = {
				variable = dap_build_vassal_holdings
				root = { add_to_variable_list = { name = dap_build_holdings target = prev } }
			}
			dap_choose_stacking_counties_effect = yes
			dap_pick_buildings_effect = yes
		}
		ordered_in_list = {
			variable = dap_build_holdings
			limit = {
				has_variable = dap_bc_value
				NOT = { has_variable = dap_build_queued }
			}
			order_by = var:dap_bc_value
			max = 100
			check_range_bounds = no
			set_local_variable = {
				name = dap_build_need
				value = {
					value = local_var:dap_build_spent
					add = var:dap_bc_cost
				}
			}
			if = {
				limit = {
					local_var:dap_build_left > 0
					root = { dap_can_spend_trigger = { AMOUNT = local_var:dap_build_need } }
				}
				change_local_variable = { name = dap_build_left subtract = 1 }
				set_local_variable = { name = dap_build_spent value = local_var:dap_build_need }
				# prev = this holding inside root
				if = {
					limit = { dap_dry_run_trigger = yes }
					root = {
						dap_log_effect = { ACTION = dap_log_build_suggested }
						set_variable = { name = dap_log_building value = prev.var:dap_bc_building }
						set_variable = { name = dap_log_province value = prev }
					}
				}
				else = {
					set_variable = { name = dap_build_queued value = 1 days = 1 }
					root = {
						add_to_variable_list = { name = dap_build_queue target = prev }
						if = {
							limit = { NOT = { has_variable = dap_build_season_count } }
							set_variable = { name = dap_build_season_count value = 0 days = 90 }
						}
						change_variable = { name = dap_build_season_count add = 1 }
					}
				}
			}
		}
	}
}
