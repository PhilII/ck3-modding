# Appends to vanilla; never adds a trigger or effect to the vanilla on_action
quarterly_playable_pulse = {
	on_actions = { dap_quarterly_pulse }
}

dap_quarterly_pulse = {
	trigger = {
		is_ai = no
		has_variable = dap_initialized
	}
	effect = {
		dap_run_all_effect = yes
	}
}

# When the player continues as their heir (or anyone else), the settings come along
on_player_character_change = {
	on_actions = { dap_player_change }
}

dap_player_change = {
	trigger = {
		exists = scope:previous_player_character
		scope:previous_player_character = { has_variable = dap_initialized }
	}
	effect = {
		dap_inherit_settings_effect = { FROM = scope:previous_player_character }
		dap_init_player_effect = yes
		# The heir's window lists their military bases at once
		dap_update_military_bases_effect = yes
	}
}
